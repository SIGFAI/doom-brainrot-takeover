		Stop;
	}
}

class BrainrotHandler : EventHandler
{
	override void PlayerEntered(PlayerEvent e)
	{
		Console.MidPrint(null, "BRAINROT TAKEOVER");
	}

	override void WorldTick()
	{
		if (level.time % 25 != 0) return;
		let pl = players[consoleplayer].mo;
		if (!pl) return;
		let r = EmojiRain(Actor.Spawn("EmojiRain", pl.Vec3Offset(frandom(-250, 250), frandom(-250, 250), 160 + frandom(0, 60))));
		if (r) r.vel.z = -5;
	}

	override void WorldThingDamaged(WorldEvent e)
	{
		if (e.Thing && e.Thing.bIsMonster && e.Damage > 0 && (e.Thing is "SharkSneakers" || e.Thing is "TungLog" || e.Thing is "EmojiSkull"))
		{
			e.Thing.A_SpawnItemEx("EmojiPop", 0, 0, e.Thing.Height * 0.6, frandom(-2, 2), frandom(-2, 2), 3, 0, SXF_NOCHECKPOSITION);
		}
	}

	override void RenderOverlay(RenderEvent e)
	{
		int t = level.time;
		if (t > 35 * 14) return;
		Font f = Font.GetFont("BigFont");
		String s = "BRAINROT TAKEOVER";
		int w = f.StringWidth(s);
		int col = (t / 5) % 2 ? Font.CR_YELLOW : Font.CR_CYAN;
		int bob = int(sin(t * 12) * 4);
		Screen.DrawText(f, col, (240 - w) / 2, 14 + bob, s, DTA_VirtualWidth, 240, DTA_VirtualHeight, 150);
		if (t < 40 && (t / 4) % 2 == 0)
			Screen.Dim(0xFF00FF, 0.25, 0, 0, Screen.GetWidth(), Screen.GetHeight());
	}
}
